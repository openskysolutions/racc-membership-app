server {
    listen 80;
    server_name members.richfieldareachamber.com richfieldareachamber.com www.richfieldareachamber.com;
    return 301 https://members.richfieldareachamber.com$request_uri;
}

# richfieldareachamber.com → members.* redirect (HTTPS)
server {
    listen 443 ssl;
    server_name richfieldareachamber.com;

    ssl_certificate_key /etc/letsencrypt/live/members.richfieldareachamber.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

# Main app (canonical domain)
server {
    listen 443 ssl;
    server_name members.richfieldareachamber.com;
    ssl_certificate /etc/letsencrypt/live/members.richfieldareachamber.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/members.richfieldareachamber.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # Allow larger request bodies for image uploads (base64 encoded adds ~33% overhead)
    client_max_body_size 15M;

    # API routes - proxy to backend service
    location /api {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
    }

    # OG image proxy - no cache-override headers so Facebook can fetch images
    location /blog-image {
        proxy_pass http://localhost:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # All other routes - proxy to frontend service for SPA routing
    location / {
        proxy_pass http://localhost:3001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;

        # Cache busting headers for development
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header Pragma "no-cache";
        add_header Expires "0";
    }
}
